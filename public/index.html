// SLS WIFI EVENTOS - hEX - index.js FINAL v13 - 07/08/2026 // Corrige 404 /api/gerar-qrcode + CORS + MercadoPago const express = require('express'); const cors = require('cors'); const crypto = require('crypto'); const app = express(); const PORT = process.env.PORT || 10000; app.use(cors({ origin: '*' })); app.use(express.json()); app.use(express.urlencoded({ extended: true })); let FILA = []; // {txid, mac, valor, tempo, plano, brcode, qrcode, status, createdAt} const MP_TOKEN = process.env.MP_ACCESS_TOKEN || process.env.MERCADOPAGO_ACCESS_TOKEN || ''; function genTxid() { return 'SLS-' + crypto.randomBytes(4).toString('hex').toUpperCase(); } // ROTA QUE ESTAVA QUEBRADA - 404 CORRIGIDO app.post('/api/gerar-qrcode', async (req, res) => { try { const { mac, valor, tempo, plano } = req.body; if (!mac) return res.status(400).json({ error: 'mac obrigatorio' }); const txid = genTxid(); const valorFinal = Number(valor) || 5; const tempoFinal = tempo || '1 hora'; const planoFinal = plano || 'SLS WIFI'; let brcode = ''; let qrcode_base64 = ''; if (MP_TOKEN) { try { const mpRes = await fetch('https://api.mercadopago.com/v1/payments', { method: 'POST', headers: { 'Authorization': `Bearer ${MP_TOKEN}`, 'Content-Type': 'application/json', 'X-Idempotency-Key': txid }, body: JSON.stringify({ transaction_amount: valorFinal, description: `${planoFinal} - ${mac}`, payment_method_id: 'pix', payer: { email: 'cliente@slswifi.com' }, external_reference: txid, notification_url: `https://hotsport-pix-2.onrender.com/api/webhook` }) }); const mpData = await mpRes.json(); if (mpData.point_of_interaction?.transaction_data) { brcode = mpData.point_of_interaction.transaction_data.qr_code; qrcode_base64 = mpData.point_of_interaction.transaction_data.qr_code_base64; } } catch (e) { console.log('MP erro, vai mock:', e.message); } } if (!brcode) { brcode = `00020126580014BR.GOV.BCB.PIX0136${txid}0215SLS WIFI EVENTOS520400005303986540${valorFinal.toFixed(2)}5802BR5920SLS WIFI EVENTOS6009TERESINA62070503***6304ABCD`; } const item = { txid, mac: mac.toUpperCase(), valor: valorFinal, tempo: tempoFinal, plano: planoFinal, brcode, qrcode: qrcode_base64, status: 'PENDENTE', createdAt: Date.now() }; FILA.push(item); setTimeout(() => { FILA = FILA.filter(f => f.txid !== txid); }, 15 * 60 * 1000); return res.json({ txid, qrcode: qrcode_base64, brcode, valor: valorFinal, tempo: tempoFinal, plano: planoFinal }); } catch (err) { res.status(500).json({ error: 'failed', details: err.message }); } }); app.get('/api/fila', (req, res) => { const { txid, mac } = req.query; if (txid) return res.json(FILA.find(f => f.txid === txid) || { status: 'NAO_ENCONTRADO' }); if (mac) return res.json(FILA.filter(f => f.mac === mac.toUpperCase())); res.json(FILA); }); app.get('/api/status/:txid', (req, res) => { res.json(FILA.find(f => f.txid === req.params.txid) || { status: 'NAO_ENCONTRADO' }); }); app.get('/api/liberacoes', (req, res) => { res.json(FILA.filter(f => f.status === 'PAGO_LIBERAR')); }); app.get('/api/liberacoes/limpar', (req, res) => { const { txid } = req.query; if (txid) FILA = FILA.filter(f => f.txid !== txid); else FILA = FILA.filter(f => f.status !== 'PAGO_LIBERAR'); res.send('LIMPO'); }); app.post('/api/webhook', async (req, res) => { try { const paymentId = req.body?.data?.id || req.query.id; if (!paymentId || !MP_TOKEN) return res.sendStatus(200); const r = await fetch(`https://api.mercadopago.com/v1/payments/${paymentId}`, { headers: { 'Authorization': `Bearer ${MP_TOKEN}` } }); const pay = await r.json(); if (pay.status === 'approved') { const item = FILA.find(f => f.txid === pay.external_reference); if (item) item.status = 'PAGO_LIBERAR'; } res.sendStatus(200); } catch { res.sendStatus(200); } }); // TESTE SEM PAGAR: https://hotsport-pix-2.onrender.com/api/pagar/SLS-XXXX app.get('/api/pagar/:txid', (req, res) => { const item = FILA.find(f => f.txid === req.params.txid); if (item) { item.status = 'PAGO_LIBERAR'; return res.json({ ok: true, item }); } res.status(404).json({ error: 'nao encontrado' }); }); app.get('/', (req, res) => res.send('SLS WIFI EVENTOS API ONLINE v13 - QR FIX')); app.listen(PORT, () => console.log('Rodando ' + PORT));
